% ---------------------------------------------------------------------------
% DOCUMENT SETTINGS, read by the shared layout below. Each is a

% ---------------------------------------------------------------------------

% The typeface: cfr-lm, as this CV has always been set. It carries the whole
% font setup rather than a font name, because the template's default selects a
% system face with fontspec and an NFSS package takes no such argument.
\newcommand{\cvtypeface}{\usepackage{cfr-lm}\usepackage[T1]{fontenc}}

% The course tables are hand-tuned: the first is 3mm wider in the second column
% than the ones below it. The generated specification is equal-width columns;
% this returns the tuned one. Tables are numbered in printing order by the
% `cvcoursetable` counter, which cv/preamble.tex steps outside the table.
\newcommand{\cvcoursecols}[1]{%
  \ifnum\value{cvcoursetable}=1 P{3.1cm} P{3.8cm} Y P{1.2cm}%
  \else P{3.1cm} P{3.5cm} Y P{1.2cm}\fi}

% This bibliography's own tidying, added to the two maps cv/preamble.tex
% declares. biblatex takes exactly one \DeclareSourcemap, so these go through
% the hook rather than through a second declaration.
%
% Prefer structured link fields for every entry: suppress howpublished whenever
% a URL or DOI is present, then suppress URL whenever a DOI is present. For
% @misc records, map publisher to the field the standard biblatex driver uses
% for the artifact's venue.
\newcommand{\cvsourcemaps}{%
  \map{
    \step[fieldsource=url, final]
    \step[fieldset=howpublished, null]
  }
  \map{
    \step[fieldsource=doi, final]
    \step[fieldset=howpublished, null]
  }
  \map{
    \step[fieldsource=doi, final]
    \step[fieldset=url, null]
  }
  \map{
    \pertype{misc}
    \step[fieldsource=publisher, fieldtarget=organization]
  }%
}

% DOCUMENT OVERRIDES: layout the template sets one way and this CV another.

% Palatino for the body, which is what this CV has always used. The template's
% \headerfontiii is \rmfamily because its own typeface is already Pagella.
\renewcommand{\headerfontiii}{\fontfamily{ppl}\selectfont} % Palatino (elegant serif)

% The entry line, set with `tabular*` rather than the template's `tabularx`: the
% left column takes its natural width and the right one is pushed to the margin,
% so a long organisation or venue name stays on one line instead of wrapping
% inside a fixed 4.8cm right column. This record has several of those, and the
% wrapping moves page breaks. \cvpartflush sets the second line the other way
% and carries its own copy of the entry, so both are renewed here.
\renewcommand{\cventry}[4]{
\vspace{0.5mm}\item
    \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
        \textbf{#1} & \textit{\footnotesize{#3}} \\
        \textit{\footnotesize{#2}} &  \footnotesize{#4}\\
    \end{tabular*}
    \vspace{-2.4mm}
}
\renewcommand{\cvpartflush}[3][\cvall]{%
  \cvdeclare{#3}%
  {%
    \renewcommand{\cventry}[4]{%
      \vspace{0.5mm}\item
      \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
          \textbf{##1} & \textit{\footnotesize{##3}} \\
          \footnotesize{\textit{##2}} & \footnotesize{##4}
      \end{tabular*}
      \vspace{-2.4mm}
    }%
    \cvpart[#1]{#2}{#3}%
  }%
}
